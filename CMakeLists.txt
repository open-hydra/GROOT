# CMake project file for 'GROOT'

cmake_minimum_required(VERSION 3.22)

# Name the code
set(solver GROOT)

project(${solver} LANGUAGES Fortran CXX C)

# Libraries requirements
# - ORION.
# Required by a range of releases with semver_dependency of cmake/SemVerDeps.cmake.
# Standalone, the release is checked out in lib/ORION, out of the build folder, and kept in deps.lock (both git ignored); -DSEMVER_UPDATE=ALL selects the newest one again.
set(GROOT_ORION_VERSION 2.1)
set(GROOT_ORION_URL https://github.com/MarcoGrossi92/ORION.git)
# - FiNeR.
# Directly pinned at one release (bump the gitlink and the tag together).
set(GROOT_FINER_TAG v2.3.0)
# FiNeR's own dependencies (cmake/FiNeRDependencies.cmake).
# FREE fetches them as FiNeR's author does (`fobis fetch`)
# BACKUP checks out the releases below, which GROOT builds and passes its tests with.
# When an upstream change breaks GROOT, set BACKUP here, or configure one build with -DGROOT_FINER_DEPS=BACKUP (kept in its cache until -UGROOT_FINER_DEPS).
if (NOT DEFINED GROOT_FINER_DEPS)
  set(GROOT_FINER_DEPS "FREE")
endif()
set(GROOT_FINER_BACKUP_PINS
  "PENF=v2.0.1"
  "StringiFor=v1.8.0"
  "FACE=v1.1.17"
  "FLAP=v2.5.3"
  "BeFoR64=v1.2.4")

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)

  set(CMAKE_VERBOSE_MAKEFILE FALSE)
  # Search path for cmake modules
  set(CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake/")

  # Parallelization options
  OPTION(USE_MPI    "Use the MPI library for parallelization" OFF)
  OPTION(USE_OPENMP "Use OpenMP for parallelization"          OFF)

  INCLUDE(${CMAKE_MODULE_PATH}/SetFortranFlags.cmake)
  INCLUDE(${CMAKE_MODULE_PATH}/SetParallelizationLibrary.cmake)

  # Set directories
  include(GNUInstallDirs)
  set(CMAKE_Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")

  # Libraries building
  # An external clone given by a relative path is taken from the source folder
  foreach(_name ORION FINER)
    if (FETCHCONTENT_SOURCE_DIR_${_name})
      get_filename_component(FETCHCONTENT_SOURCE_DIR_${_name} "${FETCHCONTENT_SOURCE_DIR_${_name}}" ABSOLUTE
                             BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    endif()
  endforeach()
  include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/SemVerDeps.cmake)
  semver_dependency(ORION GIT_REPOSITORY ${GROOT_ORION_URL} VERSION ${GROOT_ORION_VERSION} SOURCE_DIR lib/ORION)
  # FiNeR and its dependencies, the checks of both libraries, and FetchContent_MakeAvailable
  include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/AddLibraries.cmake)

  # radlib (optional) — enables the RC-SLW model (model = slw)
  # Not wired through ensure_submodule_initialized (that calls git): radlib may be a
  # manual copy. We only check that the C++ sources are present.
  option(USE_RADLIB "Enable the SLW model via the radlib library (RC-SLW)" OFF)
  if(USE_RADLIB)
    # Resolve radlib location: prefer the submodule path 'radlib', fall back to a manual
    # 'radlib-master' copy. Stored in the cache so src/lib/CMakeLists.txt can reuse it.
    set(RADLIB_PATH "" CACHE INTERNAL "radlib root directory")
    foreach(_cand "radlib" "radlib-master")
      if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/lib/third_party/${_cand}/src/c++/CMakeLists.txt")
        set(RADLIB_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lib/third_party/${_cand}" CACHE INTERNAL "radlib root directory")
        break()
      endif()
    endforeach()
    if(NOT RADLIB_PATH)
      message(FATAL_ERROR
        "USE_RADLIB=ON but radlib not found in lib/third_party/{radlib,radlib-master}\n"
        "Run: git submodule update --init --checkout lib/third_party/radlib")
    endif()
    set(_radlib_path "${RADLIB_PATH}")

    # The RC-SLW code reads ALBDF *.bin files via the compile-time macro RCSLW_DATA_DIR.
    # The repo ships only *.txt; the *.bin are generated by writebin.x (a POST_BUILD step
    # of the data/rcslw_data target) directly into that source folder. Point the macro
    # there so no 'make install' of radlib is needed.
    set(INSTALLED_RCSLW_DATA_DIR "${_radlib_path}/data/rcslw_data"
        CACHE PATH "radlib ALBDF .bin data directory" FORCE)

    # Build writebin.x and generate the .bin files (POST_BUILD custom command).
    if(NOT TARGET writebin.x)
      add_subdirectory("${_radlib_path}/data/rcslw_data"
                       "${CMAKE_BINARY_DIR}/lib/radlib/data")
    endif()

    # radlib C++ property library (target 'radlib'); consumes RCSLW_DATA_DIR.
    if(NOT TARGET radlib)
      add_subdirectory("${_radlib_path}/src/c++"
                       "${CMAKE_BINARY_DIR}/lib/radlib/c++")
    endif()

    # Ensure the ALBDF *.bin files exist before radlib is built. radlib's own POST_BUILD
    # only regenerates them when writebin.x is (re)linked, so a missing .bin with an
    # up-to-date writebin.x would not be rebuilt. Drive it via an OUTPUT rule instead:
    # the .bin become real build artifacts, regenerated whenever absent.
    if(TARGET radlib AND TARGET writebin.x)
      set(_rcslw_bin_stamp "${_radlib_path}/data/rcslw_data/h2o_p1_0.bin")
      add_custom_command(
        OUTPUT  "${_rcslw_bin_stamp}"
        COMMAND writebin.x
        DEPENDS writebin.x
        WORKING_DIRECTORY "${_radlib_path}/data/rcslw_data"
        COMMENT "Generating RC-SLW ALBDF .bin data files"
        VERBATIM)
      add_custom_target(rcslw_bin DEPENDS "${_rcslw_bin_stamp}")
      add_dependencies(radlib rcslw_bin)
    endif()
  endif()

else()

  # Part of a larger project, which gives GROOT its ORION and FiNeR.
  # When that project selected ORION with SemVerDeps, its release is checked against GROOT's requirement; nothing is added here.
  get_property(_groot_orion_selected GLOBAL PROPERTY SEMVER_ORION_SOURCE SET)
  if (_groot_orion_selected)
    include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/SemVerDeps.cmake)
    semver_dependency(ORION GIT_REPOSITORY ${GROOT_ORION_URL} VERSION ${GROOT_ORION_VERSION})
  endif()

endif()

# Build library
add_subdirectory(src/lib)

# Build application (only if this is the top-level project)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  add_subdirectory(src/app)
  add_subdirectory(test/homcube)
  add_subdirectory(test/cube_periodic)
  add_subdirectory(test/finite_cylinder)
  add_subdirectory(test/furnace_source)
  add_subdirectory(test/quadrilateral)
  add_subdirectory(test/emissivity)
  add_subdirectory(test/nozzle)
  add_subdirectory(test/vtk_io)
endif()
